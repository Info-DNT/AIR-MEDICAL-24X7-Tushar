# Not for crawling.
#
# The trailing $ anchors each rule to the end of the URL. Without it Disallow
# matches by PREFIX, so "Disallow: /air-ambulance-india" would also block
# /air-ambulance-india-to-international, which is live and listed in the
# sitemap. The explicit Allow below keeps that page reachable for any crawler
# that ignores $ but honours longest-match-wins.
#
# The .html forms 301 to these, and a trailing slash 404s, so the extensionless
# URL is the only form that needs a rule.

Disallow: /air-ambulance-to-india$
